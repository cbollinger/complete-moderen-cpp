#+LATEX: \captionof{listing}{Sort Algorithm Template}
#+BEGIN_SRC C++ :exports both :flags "-std=c++17 -Wall -Wextra -fno-elide-constructors" :results output verbatim
#include <iostream>
#include <string>

using namespace std;

template<typename T, int size>
void Sort(T(&arr)[size]) {
    for(int i = 0; i < size; ++i) {
        for(int j = 0; j < (size -1); ++j) {
            if(arr[j] > arr[j+1]) {     // will become a predicate
                T temp = arr[j];        // swap by value
                arr[j] = arr[j+1];
                arr[j+1] = temp;
            }
        }
    }
}


int main() {

    int arr[] {10, 7, 3, 128, 64, 77, 99 };

    cout << "unsorted values:\t";
    for(const auto& a : arr)
        cout << a << ", ";
    cout << endl;

    cout << "sorted values:\t";
    Sort(arr);
    for(const auto& a : arr)
        cout << a << ", ";
    cout << endl;
    
    return 0;
}
#+END_SRC

#include <iomanip>
#include <string>

// Function Prototype:
using Comperator = bool(*)(int,int);

template<typename T, int size>
void Sort(T(&arr)[size], Comperator comp) {
    for(int i = 0; i < size; ++i) {
        for(int j = 0; j < (size -1); ++j) {
            if( comp(arr[j], arr[j+1]) )    {
                T temp   = move(arr[j]); 
                arr[j]   = move(arr[j+1]);
                arr[j+1] = move(temp);
            }
        }
    }
}

// Prädicate: returns bool
bool Comp1(int x, int y) {
    return x > y;              // ascedning order
}
bool Comp2(int x, int y) {
    return x < y;              // descending order
}

    cout << "ascending order:\t";
    Sort(arr, Comp1);
    for(const auto& a : arr)

    cout << endl;
    
    cout << "descending order:\t";

*Limitiation*
- Sort is a function template and will be instantiated at compile time
- But comp is invocation of a function through its addess, that is this function cannot be *inlined*.
- Function pointers cannot be optimized by the compiler
- Function pointers point to global functions and global functions can not have a state.
- *Comperator* is therefore a callback implemented as functionn pointer.

*** Function Object
- Object with overloaded function call operator it is also know as the *functor*
- Call to overloaded function call operator resembles a local function call
- Can be used as a *callback* instead of function pointers
- More efficient than function pointers
- Usually implemented as structs

*** Function Pointer vs. Function Objects

#+ATTR_LATEX:  :align p{8cm} p{8cm}
| <40>                        | <40>                              |
| *Function Pointer*            | *Function Object*                   |
|-----------------------------+-----------------------------------|
| Invoked through a pointer   | Invoked through an object         |
| Dynamic in nature           | Static in nature                  |
| Can be specified at runtime | Must be specified at compile time |
| Difficult to optimize       | Easy to optimize (Compiler)       |
| Slow                        | Fast                              |
| Cannot store state          | Can store state                   |


** Lambda Expressions

- Defines an anounymous function object
- Syntactic shortcut for a function object
- Can be passed as an argument
- Can accpet parameters and return values
- Typically, encapsulates a few lines of code
- Behaves like a normal function when invoked
- Replaces function objects
- Internally, implemented as a nameless function object
- Use auto to provide an explicite name


#+LATEX: \captionof{listing}{Lambda Expressions}
#+BEGIN_SRC C++ :exports both :flags "-std=c++17 -Wall -Wextra -fno-elide-constructors" :results output verbatim
#include <iostream>
#include <iomanip>

using namespace std;

int main() {

    []() {
        cout  << "Welcome to Lambda Expressions 1" << endl;
    }();  // <- Use the functor to invokde the overloaded function call operator.

    auto fn = []() {  // <- give a name to the anounymous function
        cout  << "Welcome to Lambda Expressions 2" << endl;
    };  
    fn();

    cout << typeid(fn).name() << endl;
    
    return 0;
}

#+LATEX: \captionof{listing}{Lambda Expressions Capture List}
#+BEGIN_SRC C++ :exports both :flags "-std=c++17 -Wall -Wextra -fno-elide-constructors" :results output verbatim
#include <iostream>
#include <iomanip>

using namespace std;

template<typename T, int size, typename Callback> 
void ForEach(T(&arr)[size], Callback operation) {
    for(int i=0; i<size; ++i) {
        operation(arr[i]);
    }
}

int main() {

    int arr[] {10, 7, 3, 128, 64, 77, 99 };
    cout << "Input Array:\t";
    ForEach(arr, [](auto x) { cout << setw(5) << x << " ";});

    int offset = 5;
    cout << endl;
    cout << "With Offset 5:\t";
    ForEach(arr, [offset](auto &x) {x += offset;});   // offset is captured by value
    ForEach(arr, [](auto x) { cout << setw(5) << x << " ";});

    // MUTABLE REMOVES THE CONST FROM THE FUNCTOR
    cout << endl;
    cout << "Mutable Offset:\t";
    ForEach(arr, [offset](auto &x) {x -= offset;});   // set to original values
    ForEach(arr, [offset](auto &x) mutable { x += offset; ++offset; });
    ForEach(arr, [](auto x) { cout << setw(5) << x << " ";});

    return 0;
}
#+END_SRC

*** Capture List Modes

#+ATTR_LATEX:  :align p{8cm} p{8cm}
| <40>         | <40>                                                |
| *Capture Mode* | *Explanation*                                         |
|--------------+-----------------------------------------------------|
| [var]        | Capture /var/ by value                                |
| [=]          | Capture all enclosing scoope variables by value     |
| [&var]       | Capture all enclosing scoope variables by reference |
| [&,var]      | Capture /var/ by value & other variables by reference |
| [=, &var]    | Capture /var/ by reference & other variables by value |
| [this]       | Capture /this/ (captures all member variables)

class Product {

    string name;
    float price;

    Product(const string &n, float p) : name(n), price(p) {

    }

            price += taxedPrice;
        });
    }
    float GetPrice() {
        return price;
    }
    const string& GetName() {
        return name;
    }
    
};

** Generalized Capture (C++14)

- This feature allows creation of new variables in the capture clause
- The type of these variables is deduced from the type produced by the expression
- Consequently, these variables must always be initialized
- If the initializer expression is a variable, the new variable can have the same or a different name
- To create a reference, you have to use /operator &/ before the variable name

  - /[var=expression](args)/
  - /[&var=expression(args)/

using namespace std;

    int x {5};

        return y + arg;
    };

    cout << f(5) << endl;
        
    return 0;
}
