** File I/O - Part I (Basics)

- C++ provides support for file I/O through following classes
  - /ofstream/ - wite to an output stream 
  - /ifstream/ - read from an input stream
  - /fstream/ - write/read stream
- Include /fstream/ header
- All classes can be used for both text & binary I/O
- Additionally, they support modes that decide how the file is opened and operated
  - /is_open()/ to check if a stream is open

*Stream Class Typedefs*
- *ios:* contains the state that is common fo all the stream classes such as
  - File open modes
  - formatting flags
- *istream & ostream:* Used for console input-output
- *ifstream & ofstream:* Used for File input-output, whereas the *fstream*
  class is a convenient class for read-write at the same time
- *istringstream & ostringsream:* Used for reading from and writing to a buffer, whereas the *stringtream*
  class is a convenient class for read-write at the same time.


*File Open Modes*
- the following modes can be used while opening a file
- These are constants defined in std::ios_base class
- Some modes can be compined e.g. /std::ios::in | std::ios::out/ etc,
  for read write at the same time

  - /app/ seek to the end before each write operation
  - /binary/ open in binary mode
  - /in/ open for reading (default)
  - /out/ open for writing
  - /trunc/ discard file contents before openeing
  - /ate/ seek to end after open
  
  
#+LATEX: \captionof{listing}{Basic Input Output}
#+BEGIN_SRC C++ :flags "-std=c++17" :exports both :results output verbatim
#include <iostream>
#include <fstream>
#include <string>
void Write() {
    std::ofstream out;     // {"data.txt"};
    out.open("data");       // open instead of using the constructor 
    out << "Hello world" << std::endl;
    out << 10 << std::endl; 
    out.close();
}
void Read() {
    std::ifstream input;     // { "data.txt" };
    input.open("data.txt");  // open instead of using the constructor 
    std::string message;
    std::getline(input, message);
    int value{};
    input >> value;
    input.close();
    std::cout << message << ":" << value << std::endl; 
}
int main() {
    Write();
    Read();
    return 0;
}

*Stream State Flags*
- The exlamation operator ! is overloaded to check for the /failbit/
  - /goodbit/ no error /boo good()/
  - /badbit/ irrecoverable stream /boo bad()/
  - /failbit/ I/O operation failed /boo fail() [operator !]/
  - /eofbit/ end of file reached during input /bool eof()/

#+LATEX: \captionof{listing}{Error Handling}
#+BEGIN_SRC C++ :flags "-std=c++17" :exports both :results output verbatim
#include <iostream>
#include <fstream>
#include <string>
void Write() {
    std::ofstream out;     // {"data.txt"};
    out.open("data");       // open instead of using the constructor 
    out << "Hello world" << std::endl;
    out << 10 << std::endl; 
    out.close();
}
void Read() {
    std::ifstream input;     // { "data.txt" };
    input.open("data.txt");  // open instead of using the constructor
    if (input.fail()) {
        std::cout << "Could not open the file" << std::endl;
        return;
    }
    std::string message;
    std::getline(input, message);
    int value{};
    input >> value;
    input >> value;
    if (input.eof()) {
        std::cout << "End of file encountered" << std::endl;
    }
    if (input.good()) { // test with a none existing file
        std::cout << "I/O operations are successful" << std::endl;
    }
    else {
        std::cout << "Some I/O operations failed" << std::endl;
    }
    input.close();
    input.clear();                      // sets the good bit and resets all other states
    input.setstate(std::ios::failbit);  // sets failbit
    std::cout << message << ":" << value << std::endl; 
}
int main() {
    Write();
    Read();
    return 0;
}
#+END_SRC

: Some I/O operations failed

The input and output streams have a pointer internally, the pointer
represents the position at which the operation will be performed in the file.

- *Output streams:* this pointer is known as the *put pointer*.
- *Input streams:* this pointer is known as the *get pointer*.

#+LATEX: \captionof{listing}{Placing put- and get pointer}
#+BEGIN_SRC C++ :flags "-std=c++17" :exports both :results output verbatim
#include <iostream>
#include <fstream>
#include <string>
#include <filesystem>

using namespace std;
using namespace filesystem;

void Write(const filesystem::path& project) {
    current_path(project);
    
    std::ofstream out {"test.txt"};
    if(!out) {
        out << "Could not open file for writing" << endl;
        return;
    }
    string message{"C++ was invented by Bjarne"};

    // write character per character. 
    for(char ch : message) {
        out.put(ch);
    }
    out.seekp(20, ios::beg);
    out.put('#');
}
void Read(const filesystem::path& project) {
    current_path(project);

    std::ifstream input {"test.txt"};
    if (!input.is_open()) {
        cout << "Could not open the file reading" << endl;
        return;
    }
    input.seekg(10, ios::beg);
    cout << "Current position is: " << input.tellg() << endl;
    char ch{};
    while (input.get(ch)) {
        cout << ch;
    }   
}
int main() {
    path project(current_path());
    project /= "../projects/section-11/cputil";
    current_path(project);

    Write(project);
    Read(project);    
    return 0;
}

** File I/O - Part V (Binary IO)

A binary file stores data in it's equivalent binary form. If you open
a binary file in a text editor, it will contain non-human readable
characters.

#+LATEX: \captionof{listing}{Binary Files}
#+BEGIN_SRC C++ :flags "-std=c++17" :exports both :results output verbatim
#include <iostream>
#include <fstream>
#include <string>
#include <filesystem>

using namespace std;
using namespace filesystem;

int main() {
    // set current path
    path project(current_path());
    project /= "../projects/section-11/cputil";
    current_path(project);

    ofstream textstream {"data"};
    textstream << 12345678;

    ofstream binstream("binary", ios::binary | ios::out);
    int num{12345678};
    binstream.write((const char*)&num, sizeof(num));

    cout << "data and binary filed to ../projects/section-11/cputil" << endl;
    cout << "binary is stored as little endian 0xBC614E becomes 0x4E61BC" << endl;

    num = 0;
    binstream.close();
    ifstream input{ "binary", ios::binary | ios::in};
    input.read((char*)&num, sizeof(num));
    cout << num << endl;
    
    return 0;
}
#+END_SRC

#+LATEX: \captionof{listing}{Record}
#+BEGIN_SRC C++ :flags "-std=c++17" :exports both :results output verbatim
#include <iostream>
#include <fstream>
#include <string>
#include <filesystem>

using namespace std;
using namespace filesystem;

struct Record {
    int id;
    char name[10];
};

void WriteRecord(const filesystem::path &project, Record *p) {
    current_path(project);
    
    ofstream binstream{"records", ios::binary | ios::out };
    binstream.write((const char*)p, sizeof(Record));
}

Record ReadRecord(const filesystem::path &project) {
    current_path(project);

    ifstream input {"records", ios::binary | ios::in };
    Record r;
    input.read((char*)&r, sizeof(Record));
    return r;
}

int main() {
    // set current path
    path project(current_path());
    project /= "../projects/section-11/cputil";
    current_path(project);

    Record r1 {1963, "Christian"};
    WriteRecord(project, &r1);
    Record r2 = ReadRecord(project);
    
    cout << "id: " << r2.id << ", name: " << r2.name << endl;
    
    return 0;
}
#+END_SRC

#+RESULTS:
: id: 1963, name: Christian
