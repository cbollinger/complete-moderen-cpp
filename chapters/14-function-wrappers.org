
* Function Rappers (C++11)

** std::function - I

- A general-purpose function wrapper
- Can *store* any entity thac can be *invoke* like a function. and *target* that can be invoked with
- We can store *function pointers* & *function objects* inside /std::function/.
- Stored object is callted the *target*
- This target can be invoked with the same syntax through /std::function/ object
- Invocation without a target throws /std::bad_function_call/ exception

*std::function Usage*
. std::function is a class template:  /std::function<function signature> object = {callable};/

#+LATEX: \captionof{listing}{std::function Examples}
#+BEGIN_SRC C++ :exports code 
#include <functional>
int Square (int);
int *Alloc(size_t_bytes);

std::function<int(int)>        f1 = Square;                        f1(3);
std::function<int*(size_t)>    f2 = Alloc;                         f3(1024);
std::function<bool(int)>       f3 = [](int x) {return x%2 == 0;});    f3(3);
#+END_SRC

** std::bind - I
- Wrapper that can bind some arguments to a given function
- The arguments to the target function are represented using
  placeholders or directly bound to specific values.
- Placeholders are represented using constant objects of unspecified type in /std::placehlders/ namespace
  - named _1, _2,...,_N
- Can be used to invoke a function that accepts n arguments as a function with n-m arguments
- Returns a function object through which the target is invoked

*std::function Usage*
. std::bind is a class template:  /std::bind<callable, arg1, arg2,...argN>;/

#+LATEX: \captionof{listing}{std::bind Examples}
#+BEGIN_SRC C++ :exports code 
#include <functional>
int Square (int);
int Print(const char *);

auto f1 = std::bind(Square 3);     f1();
auto f2 = std::bind(Print, _1);    f2("Hello");
#+END_SRC

#include <functional>

using namespace std;
using namespace placeholders;

int Square(int x) {
    return x*x;
}

int Add(int x, int y) {
    return x + y;
}

int Subtract(int x, int y) {
    return x - y;
}

void Print(const char * p) {
    cout << p << endl;
}

struct GreaterThan {
    bool operator()(int x, int y) const {
        return x > y;
    }
};

struct Max {
    int operator()(int x, int y) const {
        return x > y ? x : y;
    }
};

int Operation(int x, int y, function<int(int,int)>  callback) {
    return callback(x,y);
}

int Accumulate(int x, int y, int z) {
    return x + y + z;
}

int main() {

    // Type correctness in C++17
    auto f4 = [](auto && x) {return Subtract(forward<decltype(x)>(x),5);}; 
    cout << f4(100) << endl;  // Subtract(100,5)
    
    // C++20 template type parameters in lambda
    auto gt = []<typename T>(T && x, T && y){return GreaterThan{}(forward<T>(y),forward<T>(x)); };
    cout << boolalpha << gt(5,7) << endl;
        
    // C++20 template type parameters in lambda
    auto acc = []<typename T>(T && x, T && y){return Accumulate(forward<T>(x),forward<T>(y),0);};
    cout << Operation(8, 5, acc) << endl;
    
    return 0;
}
