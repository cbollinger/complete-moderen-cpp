    ++(*m_pInt);

Integer Integer::operator++(int) { // a++ means a.operator++(int)
    cout << "operator++(int)" << endl;
    Integer temp(*this);
    ++(*m_pInt);
    return temp;

bool Integer::operator==(const Integer & a) const { // c==a means c.operator==(a)
    cout << "operator==(const Integer &) const" << endl;
    return *m_pInt == *a.m_pInt;
}

Integer & Integer::operator=(const Integer & a) { // c=a means c.operator=(a)
    cout << "operator=(const Integer &) const" << endl;
    if (this != &a) {            // check for self assignemnt
    delete m_pInt;
    m_pInt = new int(*a.m_pInt);
    }
    return *this;
}

Integer & Integer::operator=(Integer && a) { // c=&&a means c.operator=(&&a)
    cout << "operator=(Integer &&) const" << endl;
    if (this != &a) {
    delete m_pInt;
    m_pInt = a.m_pInt;
    a.m_pInt = nullptr;
    }
    return *this;
}

Integer Integer::operator+(const Integer & a) const { // c+a means c.operator+(a)
    cout << "operator+(const Integer &) const" << endl;
    Integer temp;
    *temp.m_pInt = *m_pInt + *a.m_pInt; // *this->m_pInt
    return temp;
}

** Make Functions

 Modern C++ discourages manual memory management. We should avoid
 using new and delete operators directly in our code, but if we want
 to create a smart pointer we have to use new to allocate memory on
 the heap.

#+LATEX: \captionof{listing}{Make Functions Shared Pointer}
#+BEGIN_SRC C++ :flags "-std=c++20 -Wall -Wextra -fno-elide-constructors" :results output verbatim
  #include <iostream>
  #include <cstdlib>
  #include <memory>
  #include <string>

  using namespace std;

  class Point {
  public:
      Point(int x, int y) {
        cout << "Point(" << x << "," << y << ")" << endl;
      }
  };
  int main() { // without using now to allocate resources for underlaying pointer
      shared_ptr<int> p1{new int{5}};

      // creates the integer object on the heap and initializes it with 5.
      auto p2 = make_shared<int>(5);
      // samrt pointer to custom type
      auto pt = make_shared<Point>(7,200); // pass the arguments to the constructor
      // Dynamic Array - REQUIRES c++20
      auto pArr = make_shared<int[]>(5);
      pArr[0] = 100;
      return 0;
  }
  #+END_SRC
